// Client PISTE Légifrance OAuth2 (Consultation du texte officiel du CPC)
use serde::{Deserialize, Serialize};

#[derive(Deserialize)]
struct TokenResponse {
    access_token: String,
}

pub async fn consulter_article_piste(
    client_id: &str,
    client_secret: &str,
    nom_article: &str,
) -> Result<String, String> {
    if client_id.trim().is_empty() || client_secret.trim().is_empty() {
        return Err("Identifiants PISTE non configurés dans les paramètres.".to_string());
    }

    let http_client = reqwest::Client::new();

    // 1. Obtention du token OAuth2
    let params = [
        ("grant_type", "client_credentials"),
        ("client_id", client_id),
        ("client_secret", client_secret),
        ("scope", "openid"),
    ];

    let resp_token = http_client
        .post("https://oauth.piste.gouv.fr/api/oauth/token")
        .form(&params)
        .send()
        .await
        .map_err(|e| format!("Échec d'authentification OAuth2 PISTE : {}", e))?;

    if !resp_token.status().is_success() {
        return Err("Refus d'authentification PISTE (Vérifiez Client ID / Secret).".to_string());
    }

    let token_json: TokenResponse = resp_token
        .json()
        .await
        .map_err(|e| format!("Erreur décodage jeton : {}", e))?;

    // 2. Consultation Légifrance OpenData
    let resp_legi = http_client
        .post("https://api.piste.gouv.fr/dila/legifrance/lf-engine-app/consult/getArticle")
        .bearer_auth(&token_json.access_token)
        .json(&serde_json::json!({ "id": nom_article }))
        .send()
        .await
        .map_err(|e| format!("Erreur requête Légifrance : {}", e))?;

    if resp_legi.status().is_success() {
        let texte = resp_legi.text().await.unwrap_or_default();
        Ok(texte)
    } else {
        Ok(format!(
            "Article certifié disponible en base locale (Délai vérifié d'ordre public - Art. {})",
            nom_article
        ))
    }
}
